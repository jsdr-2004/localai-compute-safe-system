Here's an example of how you can write pytest tests for a function that reverses a string:

```python
import pytest

def reverse_string(s):
    return s[::-1]

# Test case 1: Empty string
def test_empty_string():
    assert reverse_string("") == ""

# Test case 2: Single character
def test_single_character():
    assert reverse_string("a") == "a"

# Test case 3: Multiple characters
def test_multiple_characters():
    assert reverse_string("hello") == "olleh"

# Test case 4: String with spaces and punctuation
def test_string_with_spaces_and_punctuation():
    assert reverse_string("Hello, World!") == "!dlroW ,olleH"

# Test case 5: Non-ASCII characters
def test_non_ascii_characters():
    assert reverse_string("¡Hola!") == "!aloh¡"

# Test case 6: Large string
def test_large_string():
    large_string = "a" * 1000
    assert reverse_string(large_string) == large_string[::-1]

# Test case 7: Error handling (non-string input)
def test_error_handling_non_string_input():
    with pytest.raises(TypeError):
        reverse_string(123)

# Test case 8: Error handling (None input)
def test_error_handling_none_input():
    with pytest.raises(TypeError):
        reverse_string(None)
```

In these tests, we're covering various scenarios such as:

- Empty string
- Single character
- Multiple characters
- String with spaces and punctuation
- Non-ASCII characters
- Large string
- Error handling for non-string input (e.g., integer or float)
