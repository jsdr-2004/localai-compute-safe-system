Here are some example pytest tests for a function that reverses a string:
```python
import pytest

def reverse_string(s):
    return s[::-1]

# Test case 1: Simple reversal
def test_reverse_simple():
    assert reverse_string("hello") == "olleh"

# Test case 2: Empty string
def test_reverse_empty():
    assert reverse_string("") == ""

# Test case 3: Single character
def test_reverse_single_char():
    assert reverse_string("a") == "a"

# Test case 4: Long string
def test_reverse_long():
    assert reverse_string("abcdefghijklmnopqrstuvwxyz") == "zyxwvutsrqponmlkjihgfedcba"

# Test case 5: String with special characters
def test_reverse_special_chars():
    assert reverse_string("hello@world!") == "!dlrow olleh"

# Test case 6: Non-string input
def test_reverse_non_str():
    with pytest.raises(TypeError):
        reverse_string(123)

# Test case 7: Input that's not a string
def test_reverse_not_str():
    with pytest.raises(TypeError):
        reverse_string([1, 2, 3])

# Test case 8: Very long string (to avoid memory error)
def test_reverse_long_string():
    s = "a" * 1000
    assert reverse_string(s) == s[::-1]
```
These tests cover a variety of scenarios, including:

* A simple reversal of a short string
* An empty string
* A single character
* A long string with lowercase letters
* A string with special characters (e.g. @, !)
* A non-string input (e.g. an integer or list)
* A very long string to avoid memory errors
