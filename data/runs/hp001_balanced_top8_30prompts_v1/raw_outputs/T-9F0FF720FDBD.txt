Here are some example pytest tests for the `reverse_string` function:

```python
import pytest
from your_module import reverse_string  # Replace 'your_module' with the actual name of the module containing the function

def test_reverse_single_character():
    assert reverse_string('a') == 'a'

def test_reverse_multiple_characters():
    assert reverse_string('hello') == 'olleh'

def test_reverse_empty_string():
    assert reverse_string('') == ''

def test_reverse_string_with_punctuation():
    assert reverse_string('hello, world!') == ', world!'

def test_reverse_long_string():
    import random
    string = ''.join(random.choice('abcdefghijklmnopqrstuvwxyz') for _ in range(100))
    assert reverse_string(string) == string[::-1]

def test Reverse_string_with_non_alphabetical_characters():
    assert reverse_string('Hello, World!)') == '!World, Hello'

def test Reversing_an_integer():
    assert reverse_string(12345) == 54321

def test Reverse_string_with_negative_numbers():
    assert reverse_string(-12345) == -54321
```

Note that in Python 3.x, strings are immutable, so we can't simply modify a string like `string[::-1]`. Instead, we join the characters of the original string into a new string and then use slicing to reverse it.
